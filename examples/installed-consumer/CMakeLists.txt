cmake_minimum_required(VERSION 3.24)
project(zafiro_installed_consumer LANGUAGES CXX)

option(ZAFIRO_CONSUMER_PYTHON "Use the installed Python adapter" ON)
option(ZAFIRO_CONSUMER_JAVASCRIPT "Use the installed JavaScript adapter" ON)
option(ZAFIRO_CONSUMER_LUA "Use the installed Lua adapter" OFF)

set(_zafiro_components)
set(_zafiro_targets zafiro::zafiro)
if(ZAFIRO_CONSUMER_PYTHON)
    list(APPEND _zafiro_components python)
    list(APPEND _zafiro_targets zafiro::python)
endif()
if(ZAFIRO_CONSUMER_JAVASCRIPT)
    list(APPEND _zafiro_components javascript)
    list(APPEND _zafiro_targets zafiro::javascript)
endif()
if(ZAFIRO_CONSUMER_LUA)
    list(APPEND _zafiro_components lua)
    list(APPEND _zafiro_targets zafiro::lua)
endif()

find_package(zafiro 0.3 CONFIG REQUIRED COMPONENTS ${_zafiro_components})
add_executable(zafiro_consumer main.cpp)
target_link_libraries(zafiro_consumer PRIVATE ${_zafiro_targets})
target_compile_features(zafiro_consumer PRIVATE cxx_std_17)
if(ZAFIRO_CONSUMER_PYTHON)
    target_compile_definitions(zafiro_consumer PRIVATE ZAFIRO_CONSUMER_PYTHON)
endif()
if(ZAFIRO_CONSUMER_JAVASCRIPT)
    target_compile_definitions(zafiro_consumer PRIVATE ZAFIRO_CONSUMER_JAVASCRIPT)
endif()
if(ZAFIRO_CONSUMER_LUA)
    target_compile_definitions(zafiro_consumer PRIVATE ZAFIRO_CONSUMER_LUA)
endif()
